SETTING UP
----------------------------------------------------------------------------------------
cds init incident-management
npm install

RUN LOCALLY - cds watch:
----------------------------------------------------------------------------------------
rm incident-management.sqlite
cds deploy --to sqlite:incident-management.sqlite
cds watch

SAMPLE DATA
----------------------------------------------------------------------------------------
cds add data


PREPARE FOR PRODUCTION
----------------------------------------------------------------------------------------
cds add hana --for production
cds env requires -4 production
cds add xsuaa --for production
cds add html5-repo
npm install
cds build --production

cds add mta


CLEANUP
----------------------------------------------------------------------------------------
cf d incident-management -r -f
cf d incident-management-db-deployer -r -f
cf d incident-management-srv -r -f

cf ds incident-management-auth -f
cf ds incident-management-db -f
cf ds incident-management-html5-repo-host -f
cf ds incident-management-html5-runtime -f
