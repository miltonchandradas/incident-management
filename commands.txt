SETTING UP
----------------------------------------------------------------------------------------
cds init incident-management
npm install

RUN LOCALLY - cds watch:
----------------------------------------------------------------------------------------
rm incident-management.sqlite
cds deploy --to sqlite:incident-management.sqlite
cds watch

SAMPLE DATA
----------------------------------------------------------------------------------------
cds add data


PREPARE FOR PRODUCTION
----------------------------------------------------------------------------------------
cds add hana --for production
cds env requires -4 production
cds add xsuaa --for production
cds add html5-repo
npm install
cds build --production
